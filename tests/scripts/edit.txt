# cursor editing: type "echX abc", go left 4, delete X, insert o
echX abc{LEFT}{LEFT}{LEFT}{LEFT}{DEL}o

edit poem.txt{ENTER}Roses are red{ENTER}TI is blue{Y=}{GRAPH}{NOENTER}

# edit again, go to end of first line, append text, quit without saving via GRAPH then DEL(no)
edit poem.txt{ENTER}{2ND}{RIGHT}!!!{GRAPH}{DEL}{NOENTER}
type poem.txt
# edit and save on exit (GRAPH then ENTER=yes)
edit poem.txt{ENTER}{DOWN}{2ND}{RIGHT} too{GRAPH}{ENTER}{NOENTER}
